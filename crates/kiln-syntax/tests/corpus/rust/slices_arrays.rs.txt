//! Slices, arrays, array repeat expressions, multidimensional arrays and slice methods.

/// Rotate a slice left by `k` in place, by three reversals.
pub fn rotate_left<T>(xs: &mut [T], k: usize) {
    if xs.is_empty() {
        return;
    }
    let k = k % xs.len();
    xs[..k].reverse();
    xs[k..].reverse();
    xs.reverse();
}

/// Binary search returning the insertion point.
pub fn lower_bound(xs: &[i32], target: i32) -> usize {
    let (mut lo, mut hi) = (0, xs.len());
    while lo < hi {
        let mid = lo + (hi - lo) / 2;
        if xs[mid] < target {
            lo = mid + 1;
        } else {
            hi = mid;
        }
    }
    lo
}

/// Transpose a fixed-size matrix.
pub fn transpose<const N: usize>(m: [[i32; N]; N]) -> [[i32; N]; N] {
    let mut t = [[0; N]; N];
    for i in 0..N {
        for j in 0..N {
            t[j][i] = m[i][j];
        }
    }
    t
}

/// Prefix sums into a new vector.
pub fn prefix_sums(xs: &[u64]) -> Vec<u64> {
    let mut out = Vec::with_capacity(xs.len() + 1);
    out.push(0);
    for &x in xs {
        let last = *out.last().unwrap_or(&0);
        out.push(last + x);
    }
    out
}

/// Split a slice at the first zero.
pub fn split_at_zero(xs: &[u8]) -> (&[u8], &[u8]) {
    match xs.iter().position(|&b| b == 0) {
        Some(i) => (&xs[..i], &xs[i + 1..]),
        None => (xs, &[]),
    }
}

/// Mutable sub-slices with `split_at_mut`.
pub fn add_halves(xs: &mut [i32]) {
    let mid = xs.len() / 2;
    let (left, right) = xs.split_at_mut(mid);
    for (l, r) in left.iter_mut().zip(right.iter()) {
        *l += *r;
    }
}

/// Arrays as values: copy, compare and iterate by value.
pub fn array_values() -> ([u8; 4], bool, u32) {
    let a = [1u8, 2, 3, 4];
    let mut b = a;
    b[0] = 9;
    let same = a == [1, 2, 3, 4];
    let total: u32 = b.into_iter().map(u32::from).sum();
    (b, same, total)
}

/// Array of arrays, slices of slices and chunks.
pub fn grid_sum() -> i32 {
    let grid: [[i32; 3]; 2] = [[1, 2, 3], [4, 5, 6]];
    let rows: &[[i32; 3]] = &grid;
    let flat: Vec<i32> = rows.iter().flatten().copied().collect();
    let pairs: i32 = flat.chunks_exact(2).map(|c| c[0] * c[1]).sum();
    pairs + rows.len() as i32
}

/// Array repeat expressions with a const length.
pub const ZEROS: [u16; 16] = [0; 16];
pub static TABLE: [[u8; 2]; 3] = [[0, 1], [2, 3], [4, 5]];

/// Slicing with every range form and `get`.
pub fn slicing(xs: &[char]) -> usize {
    let a = xs.get(1..3).map_or(0, |s| s.len());
    let b = xs.get(10).is_none() as usize;
    let c = xs.first().map_or(0, |_| 1);
    let d = xs.iter().rev().take(2).count();
    let e = xs.concat_len();
    a + b + c + d + e
}

/// A trait implemented for slices.
pub trait ConcatLen {
    /// The summed UTF-8 length.
    fn concat_len(&self) -> usize;
}

impl ConcatLen for [char] {
    fn concat_len(&self) -> usize {
        self.iter().map(|c| c.len_utf8()).sum()
    }
}

/// Sorting, deduplication and retaining.
pub fn tidy(mut v: Vec<i32>) -> Vec<i32> {
    v.sort_unstable();
    v.dedup();
    v.retain(|&x| x % 3 != 0);
    v
}

/// Exercise everything.
pub fn demo() -> usize {
    let mut r = [1, 2, 3, 4, 5];
    rotate_left(&mut r, 2);
    let t = transpose([[1, 2], [3, 4]]);
    let mut h = [1, 2, 3, 4];
    add_halves(&mut h);
    let (ar, same, total) = array_values();
    let (before, after) = split_at_zero(b"ab\0cde");
    r[0] as usize
        + lower_bound(&[1, 3, 5, 7], 6)
        + t[0][1] as usize
        + prefix_sums(&[1, 2, 3]).len()
        + before.len()
        + after.len()
        + h[0] as usize
        + ar.len()
        + usize::from(same)
        + total as usize
        + grid_sum() as usize
        + ZEROS.len()
        + TABLE[2][1] as usize
        + slicing(&['a', 'é', 'c'])
        + tidy(vec![3, 1, 2, 2, 6, 5]).len()
}
