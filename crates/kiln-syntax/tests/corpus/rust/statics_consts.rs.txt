//! `const`, `static`, `const fn`, inline `const` blocks and const evaluation.

use std::sync::atomic::{AtomicUsize, Ordering};

/// A plain constant.
pub const MAX_USERS: usize = 1 << 10;

/// A constant computed by a `const fn`.
pub const FACT_10: u64 = factorial(10);

/// A static with interior mutability.
pub static CALLS: AtomicUsize = AtomicUsize::new(0);

/// A static string slice.
pub static GREETING: &str = "hello";

/// A static table of tuples.
pub static UNITS: [(&str, u64); 4] = [("B", 1), ("KiB", 1 << 10), ("MiB", 1 << 20), ("GiB", 1 << 30)];

/// Factorial at compile time.
pub const fn factorial(n: u64) -> u64 {
    match n {
        0 | 1 => 1,
        _ => n * factorial(n - 1),
    }
}

/// Greatest common divisor, usable in const contexts.
pub const fn gcd(mut a: u64, mut b: u64) -> u64 {
    while b != 0 {
        let t = b;
        b = a % b;
        a = t;
    }
    a
}

/// A const table generated with a loop in a const block.
pub const SQUARES: [u32; 16] = {
    let mut t = [0u32; 16];
    let mut i = 0;
    while i < 16 {
        t[i] = (i * i) as u32;
        i += 1;
    }
    t
};

/// A struct with an associated const and a const constructor.
#[derive(Debug, Clone, Copy, PartialEq)]
pub struct Rational {
    pub num: i64,
    pub den: i64,
}

impl Rational {
    /// One half.
    pub const HALF: Rational = Rational::new(1, 2);
    /// Zero.
    pub const ZERO: Rational = Rational { num: 0, den: 1 };

    /// Reduced on construction.
    pub const fn new(num: i64, den: i64) -> Rational {
        let g = gcd(num.unsigned_abs(), den.unsigned_abs()) as i64;
        let g = if g == 0 { 1 } else { g };
        let s = if den < 0 { -1 } else { 1 };
        Rational { num: s * num / g, den: s * den / g }
    }

    /// Add two rationals.
    pub const fn add(self, o: Rational) -> Rational {
        Rational::new(self.num * o.den + o.num * self.den, self.den * o.den)
    }
}

/// A compile-time assertion with an inline const block.
pub fn checked_len() -> usize {
    const { assert!(MAX_USERS.is_power_of_two()) };
    MAX_USERS
}

/// Inline const blocks as array repeat operands.
pub fn empties() -> [Vec<u8>; 3] {
    [const { Vec::new() }; 3]
}

/// Const generics computed from a constant expression.
pub fn buffer() -> [u8; MAX_USERS / 256] {
    [0xAA; { MAX_USERS / 256 }]
}

/// Count calls through the static atomic.
pub fn record_call() -> usize {
    CALLS.fetch_add(1, Ordering::Relaxed) + 1
}

/// Format a byte size with the static unit table.
pub fn human(bytes: u64) -> String {
    let mut best = UNITS[0];
    for &(name, size) in UNITS.iter() {
        if bytes >= size {
            best = (name, size);
        }
    }
    format!("{} {}", bytes / best.1, best.0)
}

/// Constants in patterns.
pub fn is_special(n: u64) -> bool {
    const ANSWER: u64 = 42;
    matches!(n, ANSWER | FACT_10)
}

/// Exercise everything.
pub fn demo() -> u64 {
    let r = Rational::HALF.add(Rational::new(1, -3));
    record_call();
    let calls = record_call();
    FACT_10 % 1000
        + gcd(84, 36)
        + u64::from(SQUARES[15])
        + r.num as u64
        + r.den as u64
        + u64::from(Rational::ZERO == Rational::new(0, 5))
        + checked_len() as u64
        + empties().len() as u64
        + buffer().len() as u64
        + calls as u64
        + human(5 << 20).len() as u64
        + GREETING.len() as u64
        + u64::from(is_special(42))
}
